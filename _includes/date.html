{%- comment -%} Formats a RenderCV date (YYYY, YYYY-MM or YYYY-MM-DD) as "May 2025" or "2020" {%- endcomment -%} {%- assign date = include.date | append: "" -%} {%- assign months = "January,February,March,April,May,June,July,August,September,October,November,December" | split: "," -%} {%- assign year = date | slice: 0, 4 -%} {%- if date.size > 4 -%} {%- assign month = date | slice: 5, 2 | plus: 0 | minus: 1 -%} {{ months[month] }} {{ year }} {%- else -%} {{ year }} {%- endif -%}
